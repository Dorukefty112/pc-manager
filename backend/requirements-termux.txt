# Termux (Android) için requirements.txt türevi.
#
# PyPI'de bu projenin hiçbir native bağımlılığı için Android tekerleği yok;
# pip bunları kaynaktan derlemeye çalışırsa Rust araç zinciri (maturin,
# yüzlerce crate) gerekiyor ve telefonda ya çok uzun sürüyor ya da yarıda
# kalıyor. Bu yüzden bu dosyadaki her paketin saf Python (py3-none-any)
# tekerleği olmalı; install-termux.sh pip'i --only-binary=:all: ile çalıştırır.
#
# Termux'un kendi derlediği paketlerden gelenler (pkg install, venv
# --system-site-packages ile görür; burada YAZILMAZ):
#   psutil        -> python-psutil      (PyPI kaynak paketi Android'i reddediyor)
#   cryptography  -> python-cryptography (Rust)
#   bcrypt        -> python-bcrypt       (Rust)
#   lxml          -> python-lxml         (libxml2)
#   cffi          -> python-cryptography paketinin kurulumu getirir
#
# Bilinçli olarak çıkarılanlar:
#   pydantic v2 / pydantic-core (Rust) -> pydantic v1 (saf Python); FastAPI
#       0.125.0, v1'i destekleyen son sürüm. Kodumuz yalnızca temel BaseModel
#       kullanıyor, iki sürümde de çalışıyor (CI her ikisini de test eder).
#   uvicorn[standard] -> sade uvicorn (uvloop/httptools native); WebSocket
#       desteği ayrı 'websockets' paketinden gelir.
#   ddgs (primp: Rust + BoringSSL) -> Termux'ta web araması devre dışı;
#       arama sayfası "ddgs kütüphanesi eksik" mesajı gösterir.
#   zeroconf -> Android'de mDNS yayını uygulama izni gerektirir; eşleştirme
#       QR kodu/IP ile çalışmaya devam eder.
#   pytest -> çalışma zamanında gerekmiyor.
fastapi==0.125.0
pydantic>=1.10.26,<2
uvicorn
aiofiles
websockets
python-jose[cryptography]
passlib[bcrypt]
python-multipart
httpx
beautifulsoup4
pyotp
